   164→function initialsOf(fullName: string): string {
   165→  return fullName
   166→    .split(/\s+/)
   167→    .filter(Boolean)
   168→    .map((w) => w[0])
   169→    .slice(0, 2)
   170→    .join("")
   171→    .toUpperCase();
   172→}

   178→function NavButton({
   179→  item,
   180→  active,
   181→  collapsed,
   182→  badge,
   183→  onClick,
   184→}: {
   185→  item: NavItem;
   186→  active: boolean;
   187→  collapsed: boolean;
   188→  badge?: number;
   189→  onClick: () => void;
   190→}) {
   191→  const showBadge = typeof badge === "number" && badge > 0;
   192→  return (
   193→    <button
   194→      type="button"
   195→      onClick={onClick}
   196→      title={collapsed ? item.label : undefined}
   197→      aria-current={active ? "page" : undefined}
   198→      className={cn(
   199→        "relative flex min-h-10 w-full items-center gap-3 rounded-lg px-3 text-sm font-medium outline-none transition-colors focus-visible:ring-2 focus-visible:ring-emerald-400/60",
   200→        active
   201→          ? "bg-emerald-500/15 text-emerald-300 shadow-[inset_2px_0_0_0_#34d399]"
   202→          : "text-zinc-400 hover:bg-zinc-800/70 hover:text-zinc-100",
   203→        collapsed && "justify-center px-0"
   204→      )}
   205→    >
   206→      <item.icon className="h-[18px] w-[18px] shrink-0" />
   207→      {!collapsed && <span className="min-w-0 flex-1 truncate text-left">{item.label}</span>}
   208→      {showBadge &&
   209→        (collapsed ? (
   210→          <span className="absolute right-1.5 top-1.5 h-2 w-2 rounded-full bg-rose-400 ring-2 ring-zinc-900" />
   211→        ) : (
   212→          <span className="flex h-5 min-w-5 items-center justify-center rounded-full bg-rose-500 px-1.5 text-[10px] font-bold text-white">
   213→            {badge! > 99 ? "99+" : badge}
   214→          </span>
   215→        ))}
   216→    </button>
   217→  );
   218→}

   220→function SidebarNav({
   221→  view,
   222→  user,
   223→  collapsed,
   224→  unreadCount,
   225→  onNavigate,
   226→}: {
   227→  view: AppView;
   228→  user: AuthUser;
   229→  collapsed: boolean;
   230→  unreadCount: number;
   231→  onNavigate: (view: AppView) => void;
   232→}) {
   233→  const can = useCallback(
   234→    (key: AppView) =>
   235→      user.role === "super_admin" || user.permissions?.[key] === true,
   236→    [user]
   237→  );
   238→
   239→  return (
   240→    <nav
   241→      aria-label="Main navigation"
   242→      className="flex-1 space-y-1 overflow-y-auto overflow-x-hidden px-2 py-3"
   243→    >
   244→      {NAV_SECTIONS.map((section) => {
   245→        const items = section.items.filter((i) => can(i.key));
   246→        if (items.length === 0) return null;
   247→        return (
   248→          <div key={section.label} className="pb-1">
   249→            {!collapsed && (
   250→              <p className="px-3 pb-1.5 pt-4 text-[10px] font-semibold uppercase tracking-[0.14em] text-zinc-500">
   251→                {section.label}
   252→              </p>
   253→            )}
   254→            {collapsed && <div className="mx-3 my-3 border-t border-zinc-800" />}
   255→            <div className="space-y-0.5">
   256→              {items.map((item) => (
   257→                <NavButton
   258→                  key={item.key}
   259→                  item={item}
   260→                  active={view === item.key}
   261→                  collapsed={collapsed}
   262→                  badge={item.key === "notifications" ? unreadCount : undefined}
   263→                  onClick={() => onNavigate(item.key)}
   264→                />
   265→              ))}
   266→            </div>
   267→          </div>
   268→        );
   269→      })}
   270→    </nav>
   271→  );
   272→}

   392→function NotificationsBell({ onNavigate }: { onNavigate: (view: AppView) => void }) {
   393→  const unreadCount = useAppStore((s) => s.unreadCount);
   394→  const notifVersion = useAppStore((s) => s.notifVersion);
   395→  const bumpNotifVersion = useAppStore((s) => s.bumpNotifVersion);
   396→  const [open, setOpen] = useState(false);
   397→  const [items, setItems] = useState<NotificationRecord[] | null>(null);
   398→  const [loading, setLoading] = useState(false);
   399→  const [markingAll, setMarkingAll] = useState(false);
   400→
   401→  const fetchList = useCallback(async () => {
   402→    setLoading(true);
   403→    try {
   404→      const data = await apiGet<NotificationRecord[]>("/api/notifications");
   405→      setItems(Array.isArray(data) ? data.slice(0, 8) : []);
   406→    } catch {
   407→      setItems([]);
   408→    } finally {
   409→      setLoading(false);
   410→    }
   411→  }, []);
   412→
   413→  useEffect(() => {
   414→    if (open) void fetchList();
   415→  }, [open, notifVersion, fetchList]);
   416→
   417→  const openNotification = async (n: NotificationRecord) => {
   418→    setOpen(false);
   419→    if (!n.read) {
   420→      try {
   421→        await apiPost(`/api/notifications/${n.id}/read`);
   422→        bumpNotifVersion();
   423→      } catch {
   424→        /* non-fatal */
   425→      }
   426→    }
   427→    onNavigate(notifTargetView(n.referenceType));
   428→  };
   429→
   430→  const markAllRead = async () => {
   431→    setMarkingAll(true);
   432→    try {
   433→      await apiPost("/api/notifications/read-all");
   434→      bumpNotifVersion();
   435→      await fetchList();
   436→      toast.success("All notifications marked as read");
   437→    } catch (err) {
   438→      toast.error(
   439→        err instanceof Error ? err.message : "Could not mark notifications as read"
   440→      );
   441→    } finally {
   442→      setMarkingAll(false);
   443→    }
   444→  };
   445→
   446→  const hasUnread = (items ?? []).some((n) => !n.read);
   447→
   448→  return (
   449→    <Popover open={open} onOpenChange={setOpen}>
   450→      <PopoverTrigger asChild>
   451→        <Button
   452→          variant="ghost"
   453→          size="icon"
   454→          className="relative h-10 w-10"
   455→          aria-label={`Notifications${unreadCount > 0 ? ` — ${unreadCount} unread` : ""}`}
   456→        >
   457→          <Bell className="h-[18px] w-[18px]" />
   458→          {unreadCount > 0 && (
   459→            <motion.span
   460→              key={unreadCount}
   461→              initial={{ scale: 0.5, opacity: 0 }}
   462→              animate={{ scale: 1, opacity: 1 }}
   463→              transition={{ type: "spring", stiffness: 500, damping: 22 }}
   464→              className="absolute -right-0.5 -top-0.5 flex h-4 min-w-4 items-center justify-center rounded-full bg-rose-500 px-1 text-[10px] font-bold leading-none text-white ring-2 ring-background"
   465→            >
   466→              {unreadCount > 99 ? "99+" : unreadCount}
   467→            </motion.span>
   468→          )}
   469→        </Button>
   470→      </PopoverTrigger>
   471→      <PopoverContent align="end" className="w-80 p-0 sm:w-96">
   472→        <div className="flex items-center justify-between gap-2 border-b px-4 py-3">
   473→          <p className="text-sm font-semibold">Notifications</p>
   474→          <Button
   475→            variant="ghost"
   476→            size="sm"
   477→            className="h-8 px-2 text-xs"
   478→            disabled={!hasUnread || markingAll}
   479→            onClick={markAllRead}
   480→          >
   481→            {markingAll ? (
   482→              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   483→            ) : (
   484→              <CheckCheck className="h-3.5 w-3.5" />
   485→            )}
   486→            Mark all read
   487→          </Button>
   488→        </div>
   489→
   490→        <div className="max-h-96 overflow-y-auto">
   491→          {loading && items === null ? (
   492→            <div className="space-y-3 p-4">
   493→              {[0, 1, 2].map((i) => (
   494→                <div key={i} className="flex gap-3">
   495→                  <Skeleton className="h-9 w-9 shrink-0 rounded-lg" />
   496→                  <div className="flex-1 space-y-1.5">
   497→                    <Skeleton className="h-3.5 w-3/4" />
   498→                    <Skeleton className="h-3 w-1/2" />
   499→                  </div>
   500→                </div>
   501→              ))}
   502→            </div>
   503→          ) : !items || items.length === 0 ? (
   504→            <div className="flex flex-col items-center gap-2 px-4 py-10 text-center">
   505→              <BellOff className="h-7 w-7 text-muted-foreground/60" />
   506→              <p className="text-sm font-medium">No notifications</p>
   507→              <p className="text-xs text-muted-foreground">
   508→                You&apos;re all caught up for now.
   509→              </p>
   510→            </div>
   511→          ) : (
   512→            <ul className="divide-y">
   513→              {items.map((n) => {
   514→                const meta = notifMeta(n.type);
   515→                return (
   516→                  <li key={n.id}>
   517→                    <button
   518→                      type="button"
   519→                      onClick={() => openNotification(n)}
   520→                      className={cn(
   521→                        "flex w-full items-start gap-3 px-4 py-3 text-left transition-colors hover:bg-accent/60",
   522→                        !n.read && "bg-emerald-500/[0.06]"
   523→                      )}
   524→                    >
   525→                      <span
   526→                        className={cn(
   527→                          "mt-0.5 flex h-9 w-9 shrink-0 items-center justify-center rounded-lg",
   528→                          meta.boxClass
   529→                        )}
   530→                      >
   531→                        <meta.icon className="h-4 w-4" />
   532→                      </span>
   533→                      <span className="min-w-0 flex-1">
   534→                        <span className="flex items-center gap-2">
   535→                          <span className="min-w-0 flex-1 truncate text-sm font-medium">
   536→                            {n.title}
   537→                          </span>
   538→                          {!n.read && (
   539→                            <span
   540→                              className="h-2 w-2 shrink-0 rounded-full bg-emerald-500"
   541→                              aria-label="Unread"
   542→                            />
   543→                          )}
   544→                        </span>
   545→                        <span className="mt-0.5 line-clamp-2 block text-xs leading-relaxed text-muted-foreground">
   546→                          {n.message}
   547→                        </span>
   548→                        <span className="mt-1 block text-[11px] text-muted-foreground/70">
   549→                          {safeRelative(n.createdAt)}
   550→                        </span>
   551→                      </span>
   552→                    </button>
   553→                  </li>
   554→                );
   555→              })}
   556→            </ul>
   557→          )}
   558→        </div>
   559→
   560→        <div className="border-t p-2">
   561→          <Button
   562→            variant="ghost"

   628→  const notifVersion = useAppStore((s) => s.notifVersion);
   629→
   630→  const [collapsed, setCollapsed] = useState(() => {
   631→    // lazy initializer — restores the persisted preference without an
   632→    // extra effect pass (AppShell never renders on the server)
   633→    if (typeof window === "undefined") return false;
   634→    try {
   635→      return window.localStorage.getItem(SIDEBAR_COLLAPSE_KEY) === "1";
   636→    } catch {
   637→      return false;
   638→    }
   639→  });
   640→  const [mobileOpen, setMobileOpen] = useState(false);
   641→
   642→  const toggleCollapsed = () => {
   643→    setCollapsed((c) => {
   644→      const next = !c;
   645→      try {
   646→        window.localStorage.setItem(SIDEBAR_COLLAPSE_KEY, next ? "1" : "0");
   647→      } catch {
   648→        /* ignore */
   649→      }
   650→      return next;
   651→    });
   652→  };
   653→
   654→  // ---- unread notification count: on mount, every 30s, and on refresh signal
   655→  const refreshUnread = useCallback(async () => {
   656→    try {
   657→      const res = await apiGet<{ count: number }>("/api/notifications/unread-count");
   658→      setUnreadCount(typeof res?.count === "number" ? res.count : 0);
   659→    } catch {
   660→      /* backend may be transiently unavailable — keep last known count */
   661→    }
   662→  }, [setUnreadCount]);
   663→
   664→  useEffect(() => {
   665→    void refreshUnread();
   666→  }, [refreshUnread, notifVersion]);
   667→
   668→  useEffect(() => {
   669→    const id = setInterval(() => void refreshUnread(), 30_000);
   670→    return () => clearInterval(id);
   671→  }, [refreshUnread]);
   672→
   673→  const handleNavigate = useCallback(
   674→    (target: AppView) => {
   675→      navigate(target);
   676→      setMobileOpen(false);
   677→    },
   678→    [navigate]
   679→  );
   680→
   681→  const canView =
   682→    user.role === "super_admin" || user.permissions?.[view] === true;
   683→
   684→  const ActiveView = Views[view];
   685→  const year = new Date().getFullYear();
   686→

   692→

   695→        className={cn(

   704→          collapsed={collapsed}
   705→          unreadCount={unreadCount}
   706→          onNavigate={handleNavigate}
   707→        />
   708→        <div className="shrink-0 border-t border-zinc-800 p-2">

   713→            title={collapsed ? "Expand sidebar" : "Collapse sidebar"}

   715→          >
