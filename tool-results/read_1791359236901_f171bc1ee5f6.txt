    10→// Deterministic PRNG so seeded data is stable
    11→let seedState = 42;
    12→function rand(): number {
    13→  seedState = (seedState * 1103515245 + 12345) % 2147483648;
    14→  return seedState / 2147483648;
    15→}
    16→function pick<T>(arr: T[]): T {
    17→  return arr[Math.floor(rand() * arr.length)];
    18→}
    19→function randInt(min: number, max: number): number {
    20→  return Math.floor(rand() * (max - min + 1)) + min;
    21→}
    22→
    23→function ymd(d: Date): string {
    24→  return d.toISOString().slice(0, 10);
    25→}
    26→function utc(year: number, month: number, day: number): Date {
    27→  return new Date(Date.UTC(year, month, day));
    28→}
    29→function addMonths(date: Date, months: number): Date {
    30→  const d = new Date(date);
    31→  d.setUTCMonth(d.getUTCMonth() + months);
    32→  return d;
    33→}
